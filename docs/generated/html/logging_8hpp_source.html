115 auto get_time_string() const -> std::string
116 {
117 using namespace std::chrono;
118
119 system_clock::time_point now = system_clock::now();
120 // Time zone offset
121 seconds offset = current_zone()->get_info(now).offset;
122 duration sys_time = now.time_since_epoch() + offset;
123
124 int64_t s = duration_cast<seconds>(sys_time).count() % 60;
125 int64_t m = duration_cast<minutes>(sys_time).count() % 60;
126 int64_t h = duration_cast<hours>(sys_time).count() % 24;
127
128 return std::format("{:02}:{:02}:{:02}", h, m, s);
